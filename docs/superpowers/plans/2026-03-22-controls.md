```ts
import Phaser from 'phaser';
import type { GameScene } from '../scenes/GameScene';

export type ControlScheme = 'standard' | 'trackpad';

const STORAGE_KEY = 'moonsec-controls';

let _fKey: Phaser.Input.Keyboard.Key | null = null;
let _scene: GameScene | null = null;

export function buildHint(scheme: ControlScheme): string {
  const rapid = scheme === 'trackpad' ? 'F rapid' : 'RMB rapid';
  return `A/D move  SPACE jump/jetpack  LMB turret  ${rapid}  SHIFT missile  E eject  ESC pause`;
}

export class ControlsManager {
  static init(scene: GameScene): void {
    _scene = scene;
    _fKey = scene.input.keyboard!.addKey(Phaser.Input.Keyboard.KeyCodes.F);
  }

  static getScheme(): ControlScheme {
    const stored = localStorage.getItem(STORAGE_KEY);
    return stored === 'trackpad' ? 'trackpad' : 'standard';
  }

  static setScheme(s: ControlScheme): void {
    localStorage.setItem(STORAGE_KEY, s);
  }

  static isRapidFireDown(): boolean {
    if (ControlsManager.getScheme() === 'trackpad') {
      return _fKey?.isDown ?? false;
    }
    return _scene?.input.mousePointer.rightButtonDown() ?? false;
  }
}
```

private togglePause(): void {
  this.paused = !this.paused;
  if (this.paused) {
    this.scene.pause('Game');
    this.pauseBg.setVisible(true);
    this.pauseText.setVisible(true);

    this.pauseText.setVisible(false);
    this.pauseControlsText.setVisible(false);
    this.schemeBtn.setVisible(false);

}

private buildPauseControls(scheme: ControlScheme): string {
  const rapid = scheme === 'trackpad' ? 'F           Rapid fire' : 'RMB         Rapid fire';
  return [
    'A / D       Move',
    'SPACE       Jump / Jetpack',
    'LMB         Turret',
    rapid,
    'SHIFT       Missile',
    'E           Eject / Reenter',
    'ESC         Pause / Resume',
  ].join('\n');
}

private schemeBtnText(scheme: ControlScheme): string {
  return scheme === 'standard' ? 'TRACKPAD MODE' : 'STANDARD MODE';
}
